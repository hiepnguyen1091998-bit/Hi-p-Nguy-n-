    <!-- JavaScript cho tương tác và lọc ảnh -->
    <script>
        document.addEventListener('DOMContentLoaded', () => {
            const menuButton = document.getElementById('menu-button');
            const mobileMenu = document.getElementById('mobile-menu');
            const photoCards = document.querySelectorAll('.photo-card');
            const filterButtons = document.querySelectorAll('.filter-button');
            const modal = document.getElementById('image-modal');
            const modalImage = document.getElementById('modal-image');
            const modalCaption = document.getElementById('modal-caption');

            // --- 1. Chức năng Menu Mobile ---
            menuButton.addEventListener('click', () => {
                mobileMenu.classList.toggle('hidden');
            });
            
            mobileMenu.querySelectorAll('a').forEach(link => {
                link.addEventListener('click', () => {
                    mobileMenu.classList.add('hidden');
                });
            });

            // --- 2. Chức năng Modal Xem Ảnh ---
            photoCards.forEach(card => {
                card.addEventListener('click', () => {
                    // Check if the card is hidden before opening the modal
                    if (card.style.display === 'none') return;
                    
                    const imgSrc = card.getAttribute('data-src');
                    const captionText = card.getAttribute('data-caption');
                    
                    modalImage.src = imgSrc;
                    modalCaption.textContent = captionText;
                    modal.classList.remove('hidden');
                });
            });

            // Hàm đóng Modal
            window.closeModal = () => {
                modal.classList.add('hidden');
            };

            // Đóng modal bằng phím ESC
            document.addEventListener('keydown', (e) => {
                if (e.key === "Escape" && !modal.classList.contains('hidden')) {
                    closeModal();
                }
            });

            // --- 3. Chức năng Lọc Ảnh ---
            filterButtons.forEach(button => {
                button.addEventListener('click', () => {
                    const filterValue = button.getAttribute('data-filter');

                    // Cập nhật trạng thái nút (Active class)
                    filterButtons.forEach(btn => btn.classList.remove('active', 'bg-primary-dark', 'text-background-light', 'hover:bg-secondary-gold', 'hover:text-text-dark', 'bg-white', 'text-text-dark', 'border', 'border-gray-300'));

                    // Đặt lại màu sắc mặc định cho tất cả các nút
                    filterButtons.forEach(btn => {
                        btn.classList.add('bg-white', 'text-text-dark', 'border', 'border-gray-300', 'hover:bg-gray-100');
                    });

                    // Đặt màu sắc cho nút đang hoạt động
                    button.classList.add('active', 'bg-primary-dark', 'text-background-light', 'hover:bg-primary-dark');
                    button.classList.remove('bg-white', 'text-text-dark', 'border', 'border-gray-300', 'hover:bg-gray-100');


                    // Lọc và hiển thị ảnh
                    photoCards.forEach(card => {
                        const cardCategory = card.getAttribute('data-category');
                        
                        if (filterValue === 'all' || cardCategory === filterValue) {
                            // Hiển thị (sử dụng Tailwind flex/grid)
                            card.style.display = 'block';
                        } else {
                            // Ẩn
                            card.style.display = 'none';
                        }
                    });
                });
            });
        });
    </script>
